-- Create the artist schema if it doesn't exist
CREATE SCHEMA IF NOT EXISTS artist;

-- Set the search path so that the following tables are created inside artist schema
SET search_path TO artist;

-- =========================
-- Create the artists table
-- =========================
CREATE TABLE IF NOT EXISTS artists (
    artist_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    password_hash VARCHAR(255),           -- null if Google account
    bio TEXT,
    country VARCHAR(100),
    profile_image_url TEXT,
    auth_provider VARCHAR(50) DEFAULT 'local',  -- 'local' or 'google'
    provider_id VARCHAR(255),                  -- Google user ID
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =========================
-- Auto-update trigger for updated_at
-- =========================
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE TRIGGER artists_updated_at
BEFORE UPDATE ON artists
FOR EACH ROW
EXECUTE FUNCTION update_updated_at_column();

-- =========================
-- Create artist_sessions table
-- =========================
CREATE TABLE IF NOT EXISTS artist_sessions (
    session_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    artist_id INT REFERENCES artists(artist_id) ON DELETE CASCADE,
    refresh_token VARCHAR(255),
    expires_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
